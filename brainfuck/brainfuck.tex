\def\zybrf@{%
	\edef\zybrf@cell@copy{\zybrf@cell}%
	\catcode32=12
	\catcode`\^^I=12
	\ifx[\zybrf@tmp\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
		{\zybrf@@}
		{\zybrf@@[]}%
}
\def\zybrf@@[#1]#2{%
	\edef\zybrf@tmp{#1}%
	\catcode32=10
	\catcode`\^^I=10
	\zybrf@p{#2}%
	\endgroup
}
\def\zybrf@p#1{%
	\zybrf@p@\relax\zybrf@relax#1\zybrf@relax
}
\def\zybrf@p@#1\zybrf@relax#2{%
	\ifx\zybrf@relax#2\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
		{\zybrf@r{#1}}
		{\ifx]#2\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
			{\zybrf@p@@\zybrf@relax#1\zybrf@relax}
			{\zybrf@p@{#2}#1\zybrf@relax}}%
}
\def\zybrf@p@@#1\zybrf@relax#2#3\zybrf@relax{%
	\ifx\zybrf@relax#3\zybrf@relax\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
		{\zybrf@p\relax\zybrf@relax#1\zybrf@relax}
		{\ifx[#2\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
			{\zybrf@p@@@\zybrf@loop{#1}\zybrf@endloop\zybrf@relax#3\zybrf@relax}
			{\zybrf@p@@{#2}#1\zybrf@relax#3\zybrf@relax}}%
}
\def\zybrf@p@@@#1\zybrf@relax#2#3\zybrf@relax{%
	\ifx\zybrf@relax#3\zybrf@relax\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
		{\zybrf@p@\relax\zybrf@relax#1}
		{\zybrf@p@@@{#2}#1\zybrf@relax#3\zybrf@relax}%
}
\def\zybrf@r#1{%

}
\def\zybrf@r@#1#2\zybrf@relax#3\zybrf@relax{%

		{\zybrf@c#3\zybrf@relax}
		{\zybrf@r@#2\zybrf@relax{#1}#3\zybrf@relax}%
}
\def\zybrf@c#1\relax\zybrf@relax{%
